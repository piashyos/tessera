# Folders Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users organize notes into folders — create/rename/delete folders, assign a note to a folder, filter the list by folder, and show the selected folder's name in the header.

**Architecture (decisions locked):** Flat folders (no nesting). A note belongs to **exactly one** folder or none (`folderId: String?`; `null` = "All Notes" / unfiled). Assignment happens via a folder picker in the editor and a "Move to folder" swipe/sheet action on list cards. The list header shows the selected folder name (defaults to "All Notes"). Clean architecture mirrors the existing notes stack: Room entity → DAO → mapper → repository → use case → ViewModel → Compose.

**Tech Stack:** Room 2.6.1 (schema bump v1→v2 + Migration), Hilt (`@Binds` in `RepositoryModule`), Kotlin Flow, JUnit5, Turbine, Robolectric in-memory Room, Jetpack Compose.

**Depends on:** Plan 1 (`2026-06-15-notes-list-visual-polish.md`) Task 5.2 — the collapsing header already takes a `title: String` parameter. This plan supplies the folder name to it. If Plan 1 is not yet done, the header title plumbing in Chunk 5 must be added here instead.

**Conventions:** Same as the other plans. Use cases are thin `@Inject`-constructed wrappers with `operator fun invoke(...)` (see `ObserveNotesUseCase`). Repositories are `@Binds`-bound singletons. Commit after every green step. DRY, YAGNI, TDD.

// app/src/main/java/app/tessera/data/local/dao/FolderDao.kt
package app.tessera.data.local.dao

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.OnConflictStrategy
import androidx.room.Query
import app.tessera.data.local.entity.FolderEntity
import kotlinx.coroutines.flow.Flow

@Dao
interface FolderDao {
    @Query("SELECT * FROM folders ORDER BY name")
    fun observeAll(): Flow<List<FolderEntity>>

    @Query("SELECT * FROM folders WHERE id = :id")
    suspend fun getById(id: String): FolderEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun upsert(folder: FolderEntity)

    @Query("UPDATE folders SET name = :name, updatedAt = :ts WHERE id = :id")
    suspend fun rename(id: String, name: String, ts: Long)

    @Query("DELETE FROM folders WHERE id = :id")
    suspend fun delete(id: String)

    @Query("UPDATE notes SET folderId = NULL, updatedAt = :ts, syncedAt = NULL WHERE folderId = :folderId")
    suspend fun unassignNotesFromFolder(folderId: String, ts: Long)

    @Query("UPDATE notes SET folderId = :folderId, updatedAt = :ts, syncedAt = NULL WHERE id = :noteId")
    suspend fun setNoteFolder(noteId: String, folderId: String?, ts: Long)
}
```

import androidx.room.migration.Migration
import androidx.sqlite.db.SupportSQLiteDatabase

val MIGRATION_1_2 = object : Migration(1, 2) {
    override fun migrate(db: SupportSQLiteDatabase) {
        db.execSQL("ALTER TABLE notes ADD COLUMN folderId TEXT")
        db.execSQL("CREATE INDEX IF NOT EXISTS index_notes_folderId ON notes(folderId)")
        db.execSQL(
            "CREATE TABLE IF NOT EXISTS folders (" +

                "createdAt INTEGER NOT NULL, updatedAt INTEGER NOT NULL)",
        )
        db.execSQL("CREATE UNIQUE INDEX IF NOT EXISTS index_folders_name ON folders(name)")
    }
}

// app/src/main/java/app/tessera/data/repository/FoldersRepositoryImpl.kt
package app.tessera.data.repository

import app.tessera.core.util.Clock
import app.tessera.data.local.dao.FolderDao
import app.tessera.data.mapper.FolderMapper
import app.tessera.domain.model.Folder
import app.tessera.domain.model.FolderId
import app.tessera.domain.model.NoteId
import app.tessera.domain.repository.FoldersRepository
import java.util.UUID
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

@Singleton
class FoldersRepositoryImpl @Inject constructor(
    private val folderDao: FolderDao,
    private val folderMapper: FolderMapper,
    private val clock: Clock,
) : FoldersRepository {
    override fun observeFolders(): Flow<List<Folder>> =
        folderDao.observeAll().map { list -> list.map(folderMapper::toDomain) }

    override suspend fun createFolder(name: String): Folder {
        val now = clock.nowMs()
        val folder = Folder(FolderId(UUID.randomUUID().toString()), name.trim(), now, now)
        folderDao.upsert(folderMapper.toEntity(folder))
        return folder
    }

    override suspend fun renameFolder(id: FolderId, name: String) =
        folderDao.rename(id.value, name.trim(), clock.nowMs())

    override suspend fun deleteFolder(id: FolderId) {
        folderDao.unassignNotesFromFolder(id.value, clock.nowMs())
        folderDao.delete(id.value)
    }

    override suspend fun moveNoteToFolder(noteId: NoteId, folderId: FolderId?) =
        folderDao.setNoteFolder(noteId.value, folderId?.value, clock.nowMs())
}
```
